% Minimalne ustawienia pracy dyplomowej
% Zostawione głównie: marginesy, font, język, interlinia, nagłówki, podpisy, bibliografia i linki.

% Marginesy strony (druk dwustronny, lustrzane odbicie zgodnie z wymogami
% wydziału: wewnętrzny/oprawowy 3,5 cm, zewnętrzny 2,5 cm).
\usepackage[
  top=2.5cm,
  bottom=2.5cm,
  inner=3.5cm,
  outer=2.5cm,
  a4paper,
  twoside
]{geometry}

% Obsługa grafik.
\usepackage{graphicx}

% Język polski: dzielenie wyrazów, podpisy, nazwy elementów dokumentu.
\usepackage[polish]{babel}

% CZCIONKA GLOBALNA: Times New Roman (wymóg wydziału), ładowana jako prawdziwy
% font systemowy przez fontspec — dokument kompilujemy XeLaTeX-em lub LuaLaTeX-em,
% nie pdfLaTeX-em (fontenc/inputenc nie są tu potrzebne).
\usepackage{fontspec}
\setmainfont{Times New Roman}

% Mikrotypografia (protrusion/expansion) — bez tego zmiana fontu na Times New
% Roman zostawia sporo linii lekko wychodzących za margines.
\usepackage{microtype}

% Zapasowa elastyczność łamania wierszy — priorytet ma niewychodzenie tekstu
% za margines (nawet kosztem odstępów) nad idealnie równym prawym marginesem.
\tolerance=1500
\emergencystretch=3em

\usepackage[indentafter]{titlesec}

% Rozdział (nagłówek 1. stopnia): 12 pt, wersaliki, pogrubienie.
% \MakeUppercase jako before-code (5. argument) — tak, żeby uwielkościł
% wyłącznie tekst tytułu, a nie numer rozdziału z pierwszego argumentu.
\titleformat{\chapter}[block]
  {\normalfont\normalsize\bfseries}
  {\thechapter.}{0.5em}{\MakeUppercase}
\titlespacing*{\chapter}{0pt}{12pt}{6pt}

% Podrozdział (nagłówek 2. stopnia): 10 pt, pogrubienie i kursywa.
\titleformat{\section}[block]
  {\normalfont\fontsize{10pt}{12pt}\selectfont\bfseries\itshape}
  {\thesection.}{0.5em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}

% Punkt podrozdziału (nagłówek 3. stopnia): 10 pt, kursywa (bez pogrubienia).
\titleformat{\subsection}[block]
  {\normalfont\fontsize{10pt}{12pt}\selectfont\itshape}
  {\thesubsection.}{0.5em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}

% Nagłówek 4. stopnia: 10 pt, pogrubienie.
\titleformat{\subsubsection}[block]
  {\normalfont\fontsize{10pt}{12pt}\selectfont\bfseries}
  {\thesubsubsection.}{0.5em}{}
\titlespacing*{\subsubsection}{0pt}{12pt}{6pt}

% Formatowanie tytułu spisu treści i jego zawartości.
\usepackage{tocloft}
\renewcommand{\cfttoctitlefont}{\normalfont\large\bfseries\MakeUppercase}
\renewcommand{\cftchapfont}{\normalfont\bfseries}
\renewcommand{\cftchappagefont}{\normalfont\bfseries}
\renewcommand{\cftsecfont}{\normalfont}
\renewcommand{\cftsecpagefont}{\normalfont}
\renewcommand{\cftsubsecfont}{\normalfont}
\renewcommand{\cftsubsecpagefont}{\normalfont}

% Interlinia, np. 1.5.
\usepackage{setspace}

% Wcięcie pierwszego akapitu po nagłówku.
\usepackage{indentfirst}

% Estetyczne tabele: \toprule, \midrule, \bottomrule.
\usepackage{booktabs}

% Wyśrodkowane kolumny o zadanej szerokości (>{\centering\arraybackslash}p{...}).
\usepackage{array}

% Tabele przechodzące przez łamanie strony (tabela 20 cech FEATURES).
\usepackage{longtable}

% Treść wszystkich tabel składana czcionką 10 pt (wymóg wydziału).
\usepackage{etoolbox}
\AtBeginEnvironment{tabular}{\footnotesize}

% Formatowanie podpisów rysunków i tabel (usunięto kursywę dla czystości inżynierskiej).
\usepackage[
  format=plain,
  font=footnotesize,
  labelfont=bf,
  textfont=normalfont,
  labelsep=period,
  justification=centering
]{caption}

% Podstawowe środowiska i symbole matematyczne.
\usepackage{amsmath}
\usepackage{amssymb}

% Punktor list jako pik — czysto kosmetyczny akcent pokerowy (pełny czarny
% glif z amssymb, nie emoji).
\renewcommand{\labelitemi}{$\spadesuit$}

% Nagłówki i stopki.
\usepackage{fancyhdr}

% Bibliografia.
% citestyle=numeric-comp (zamiast domyślnego dla stylu ieee) — łączy wiele
% cytowań w jeden nawias kwadratowy, np. [2,3,4] / [2-4], zgodnie z wytycznymi
% wydziałowymi (zamiast oddzielnych nawiasów [2], [3], [4] ze stylu IEEE).
\usepackage[style=ieee,citestyle=numeric-comp]{biblatex}
\usepackage{csquotes}

% Linki w PDF.
% hyperref najlepiej ładować pod koniec ustawień.
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  citecolor=black,
  filecolor=black,
  linkcolor=black,
  urlcolor=black
}

% Ścieżka do grafik.
\graphicspath{{./images/}}

% Plik bibliografii.
\addbibresource{bibliografia.bib}

% Numerowanie rysunków, tabel i wzorów w obrębie rozdziału, np. 3.1, 3.2.
\counterwithin{figure}{chapter}
\counterwithin{table}{chapter}
\counterwithin{equation}{chapter}

% Polska, krótka nazwa podpisu rysunku: "Rys." zamiast "Rysunek".
\addto\captionspolish{\renewcommand{\figurename}{Rys.}}

% Stopka z numerem strony na środku.
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\fontsize{10}{12}\selectfont\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% Ten sam styl numeracji dla stron początkowych rozdziałów.
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\fontsize{10}{12}\selectfont\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% Wcięcie akapitu, brak dodatkowego odstępu między akapitami.
\setlength{\parindent}{1.25cm}
\setlength{\parskip}{0pt}

% Interlinia 1.5.
\onehalfspacing

% Pusta strona bez numeru — przydatne przy druku dwustronnym.
\newcommand\blankpage{%
  \null
  \thispagestyle{empty}%
  \addtocounter{page}{-1}%
  \newpage
}
% Obsługa kolorów i roboczych znaczników (TODO, FIXME, NOTE).
\usepackage{xcolor}

% Kolory znaczników.
\definecolor{todoYellow}{HTML}{FFDE59}
\definecolor{fixmePurple}{HTML}{A32CC4}
\definecolor{noteRed}{HTML}{FF4D4D}

% Przełącznik trybu roboczego:
%   \draftnotestrue  -- znaczniki widoczne (wersja lokalna),
%   \draftnotesfalse -- znaczniki ukryte (wersja dla promotora).
\newif\ifdraftnotes
\draftnotesfalse

% Znacznik = krótka kolorowa etykieta oraz treść płynąca jak zwykły tekst,
% dzięki czemu długie uwagi zawijają się i nie wychodzą poza margines.
% #1 kolor etykiety, #2 nazwa etykiety, #3 treść.
\newcommand{\draftmark}[3]{%
  \ifdraftnotes
    \colorbox{#1}{\sffamily\bfseries\footnotesize #2}%
    \nobreakspace\textcolor{#1!75!black}{#3}%
  \fi
}
\newcommand{\TODO}[1]{\draftmark{todoYellow}{TODO}{#1}}
\newcommand{\FIXME}[1]{\draftmark{fixmePurple}{FIXME}{#1}}
\newcommand{\NOTE}[1]{\draftmark{noteRed}{NOTE}{#1}}

\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta}
